Q4

Team Avg Performance Score

⌥ SQL
SELECT TOP 10 e.Manager_Id, e.Manager_Name,
  COUNT(DISTINCT e.Employee_Id) AS team_size,
  ROUND(AVG(CAST(mp.Performance_Rating AS FLOAT)), 3) AS avg_team_perf

WHERE e.Manager_Id IS NOT NULL

HAVING COUNT(DISTINCT e.Employee_Id) >= 3

Q5
Đào tạo & Hiệu suất
Nhân viên học nhiều hơn có hiệu suất cao hơn không?
Avg Performance by Training Hours
Avg Satisfaction by Training Hours
Insight: Ngưỡng tối ưu là 5–9h/tháng (perf 3.939, sat 7.529). Trên 15h — "training overload effect" — không mang lại lợi ích thêm.
⌥ SQL
WITH bucketed AS (
  SELECT
    CASE
      WHEN CAST(Training_Hours AS FLOAT) < 5  THEN '0-4h'
      WHEN CAST(Training_Hours AS FLOAT) < 10 THEN '5-9h'
      WHEN CAST(Training_Hours AS FLOAT) < 15 THEN '10-14h'
      WHEN CAST(Training_Hours AS FLOAT) < 20 THEN '15-19h'
      ELSE '20h+'
    END AS training_bucket,
    CAST(Performance_Rating AS FLOAT) AS perf,
    CAST(Employee_Satisfaction AS FLOAT) AS satisfaction
  FROM dbo.MonthlyPerformance
)
SELECT training_bucket,
  ROUND(AVG(perf), 3) AS avg_perf,
  ROUND(AVG(satisfaction), 3) AS avg_satisfaction
FROM bucketed
GROUP BY training_bucket
ORDER BY training_bucket;

Q9
Ứng viên tiềm năng được thăng tiến
Weighted score: Performance×0.4 + Productivity×0.5 + Satisfaction×0.1
#NameDepartmentRoleLevelPerfProd.Sat.Score
Insight: Anna Wheeler (Shift Supervisor) dẫn đầu (3.579). 8/15 ứng viên từ Store Operations — bộ phận lớn nhất cũng nhiều tài năng nhất.
⌥ SQL
SELECT TOP 15 e.Employee_Id, e.Full_Name, e.Department, e.Job_Role, e.Job_Level,
  ROUND(
    AVG(CAST(mp.Performance_Rating AS FLOAT)) * 0.4 +
    AVG(CAST(mp.Employee_Satisfaction AS FLOAT)) * 0.1 +
    AVG(CAST(rk.Productivity_Index AS FLOAT)) * 0.5
  , 3) AS promotion_score
FROM dbo.Employees e
JOIN dbo.MonthlyPerformance mp ON e.Employee_Id = mp.Employee_Id
JOIN dbo.RoleKPIs rk ON e.Employee_Id = rk.Employee_Id AND mp.Year_Month = rk.Year_Month
WHERE e.Exit_Date IS NULL AND e.Job_Level != 'Executive'
GROUP BY e.Employee_Id, e.Full_Name, e.Department, e.Job_Role, e.Job_Level
HAVING COUNT(*) >= 6
ORDER BY promotion_score DESC;

WITH age_groups AS (
  SELECT
    CASE
      WHEN e.Age < 25 THEN 'Under 25'
      WHEN e.Age < 30 THEN '25-29'
      WHEN e.Age < 35 THEN '30-34'
      WHEN e.Age < 40 THEN '35-39'

    CAST(mp.Performance_Rating AS FLOAT) AS perf
  FROM dbo.Employees e
  JOIN dbo.MonthlyPerformance mp ON e.Employee_Id = mp.Employee_Id
)
SELECT age_group, ROUND(AVG(perf), 3) AS avg_perf
FROM age_groups
GROUP BY age_group
ORDER BY MIN(/* original age */...);

⌥ SQL
WITH tenured AS (
  SELECT DATEDIFF(MONTH,
      TRY_CONVERT(DATE, Hire_Date, 103),
      TRY_CONVERT(DATE, Exit_Date, 103)) AS tenure_months
  FROM dbo.Employees
  WHERE Exit_Date IS NOT NULL AND
        TRY_CONVERT(DATE, Hire_Date, 103) IS NOT NULL
)
SELECT
  CASE
    WHEN tenure_months < 6  THEN '< 6 months'
    WHEN tenure_months < 12 THEN '6-12 months'
    WHEN tenure_months < 24 THEN '1-2 years'
    WHEN tenure_months < 36 THEN '2-3 years'
    ELSE '3+ years'
  END AS tenure_bucket, COUNT(*) AS exited
FROM tenured
GROUP BY CASE ... END
ORDER BY MIN(tenure_months);
RISK
Flight Risk Score — Nhân viên có nguy cơ nghỉ cao nhất
Score = Absenteeism×0.3 + (10–Satisfaction)×0.4 + (5–Performance)×0.3
#Nhân viênDepartmentAbsentSat.Perf.Risk Score
Insight: Logistics & IT có mật độ nhân viên nguy cơ cao nhất. HR nên review danh sách này hàng tháng và tiếp cận trong vòng 1–3 tháng trước khi họ quyết định nghỉ.
⌥ SQL
SELECT TOP 15 e.Full_Name, e.Department, e.Job_Role,
  ROUND(AVG(CAST(mp.Absenteeism_Days AS FLOAT)), 2) AS avg_absent,
  ROUND(AVG(CAST(mp.Employee_Satisfaction AS FLOAT)), 2) AS avg_sat,
  ROUND(
    AVG(CAST(mp.Absenteeism_Days AS FLOAT)) * 0.3 +
    (10 - AVG(CAST(mp.Employee_Satisfaction AS FLOAT))) * 0.4 +
    (5  - AVG(CAST(mp.Performance_Rating AS FLOAT))) * 0.3
  , 2) AS flight_risk_score
FROM dbo.Employees e
JOIN dbo.MonthlyPerformance mp ON e.Employee_Id = mp.Employee_Id
WHERE e.Exit_Date IS NULL
GROUP BY e.Employee_Id, e.Full_Name, e.Department, e.Job_Role, e.Job_Level
HAVING COUNT(*) >= 6
ORDER BY flight_risk_score DESC;
